# Editorial Governance Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the current CMS foundation into a handoff-safe editorial surface by introducing a clearer role model, separating publishing from system-wide controls, adding workflow guardrails, and exposing operator-readable audit history.

**Architecture:** This plan only covers Track 1 from the approved architecture spec. It keeps the existing Payload collections and draft system, but adds a stricter editorial layer around them: clearer roles and capability defaults, a dedicated system-mode global, workflow metadata on publishable content, and a read-only operational audit collection generated by hooks.

**Tech Stack:** Payload CMS 3, Next.js App Router, TypeScript, Vitest, Markdown docs

## Global Constraints

- single-site
- marketing-focused
- editorial-first
- clone-oriented for future client projects
- safe for non-technical operators after handoff
- content can move through `draft`, `in_review`, `scheduled`, `published`, and `reverted` states
- users with editing rights do not automatically inherit publish or system-mode control
- this does not require a heavyweight enterprise workflow engine
- workflow safety first, block breadth second
- this phase does not include multi-tenant architecture
- this phase does not include generalized support for every hosting or database combination
- this phase does not include a broad block-library expansion program
- this phase does not include a generic abstraction layer for all possible CMS use cases

export const editorialRoleOptions = [
  {
    label: 'Admin',
    value: 'admin',
  },
  {
    label: 'Publisher',
    value: 'publisher',
  },
  {
    label: 'Editor',
    value: 'editor',
  },
  {
    label: 'Marketing Ops',
    value: 'marketingOps',
  },
] as const

export type EditorialRole = (typeof editorialRoleOptions)[number]['value']

const validRoles = new Set<EditorialRole>(editorialRoleOptions.map(({ value }) => value))

export const isEditorialRole = (value: unknown): value is EditorialRole => {
  return typeof value === 'string' && validRoles.has(value as EditorialRole)
}

export const normalizeEditorialRole = (
  value: unknown,
  fallback: EditorialRole = 'editor',
): EditorialRole => {
  return isEditorialRole(value) ? value : fallback
}

import type { CollectionBeforeChangeHook, CollectionConfig } from 'payload'

import { editorialRoleOptions } from '@/access/editorialRoles'
import {
  adminCapabilityCheckboxFields,
  adminOnlyAccess,
  adminOrFirstUserAccess,
  allAdminCapabilities,
  getDefaultCapabilitiesForRole,
  hideFromNonAdmin,
  normalizeAdminRole,
  toAdminCapabilitiesMap,
} from '@/access/adminCapabilities'

const showCapabilitiesForNonAdminRole = (
  data: Record<string, unknown>,
  siblingData: Record<string, unknown>,
  { user }: { user?: unknown },
) => {
  if (!user) {
    return false
  }

  const resolvedRole = normalizeAdminRole(siblingData?.role ?? data?.role, 'editor')

  return resolvedRole !== 'admin'
}

const normalizeUserAccessFields: CollectionBeforeChangeHook = async ({
  data,
  operation,
  originalDoc,
  req,
}) => {
  if (!data) {
    return data
  }

  if (operation === 'create') {
    const { totalDocs } = await req.payload.count({
      collection: 'users',
      overrideAccess: true,
      where: {},
    })

    if (totalDocs === 0) {
      data.role = 'admin'
      data.capabilities = toAdminCapabilitiesMap(allAdminCapabilities)
      return data
    }
  }

  const defaultRole = normalizeAdminRole(originalDoc?.role, 'editor')
  const nextRole = normalizeAdminRole(data.role, defaultRole)

  data.role = nextRole
  data.capabilities =
    nextRole === 'admin'
      ? toAdminCapabilitiesMap(allAdminCapabilities)
      : {
          ...getDefaultCapabilitiesForRole(nextRole),
          ...toAdminCapabilitiesMap(data.capabilities ?? originalDoc?.capabilities),
        }

  return data
}

export const Users: CollectionConfig = {
  slug: 'users',
  admin: {
    group: 'Admin',
    hidden: hideFromNonAdmin,
    useAsTitle: 'email',
  },
  auth: true,
  access: {
    create: adminOrFirstUserAccess,
    delete: adminOnlyAccess,
    read: adminOnlyAccess,
    readVersions: adminOnlyAccess,
    update: adminOnlyAccess,
  },
  hooks: {
    beforeChange: [normalizeUserAccessFields],
  },
  fields: [
    {
      name: 'role',
      type: 'select',
      defaultValue: 'editor',
      options: [...editorialRoleOptions],
      required: true,
      saveToJWT: true,
    },
    {
      name: 'capabilities',
      type: 'group',
      saveToJWT: true,
      admin: {
        condition: showCapabilitiesForNonAdminRole,
        description:
          'Choose which admin areas this user can access. Admin users always receive full access.',
      },
      fields: [...adminCapabilityCheckboxFields],
    },
  ],
}

import type { GlobalConfig, RelationshipFieldSingleValidation } from 'payload'

import {
  allowPublicReadOrCapability,
  hideFromUsersWithoutCapability,
  requireCapability,
} from '@/access/adminCapabilities'
import type { SystemPageType } from '@/lib/systemPageTypes'

const extractPageID = (
  value: number | { id: number | string } | null | undefined,
): number | string | null => {
  if (typeof value === 'number' || typeof value === 'string') {
    return value
  }

  if (typeof value === 'object' && value !== null) {
    return value.id
  }

  return null
}

const validatePublishedSystemPageType =
  (expectedType: SystemPageType, fieldLabel: string): RelationshipFieldSingleValidation =>
  async (value, { req }) => {
    const pageID = extractPageID(value as number | { id: number | string } | null | undefined)

    if (pageID == null) {
      return true
    }

    const result = await req.payload.find({
      collection: 'system-pages',
      where: {
        and: [
          { id: { equals: pageID } },
          { _status: { equals: 'published' } },
          { type: { equals: expectedType } },
        ],
      },
      depth: 0,
      limit: 1,
      req,
    })

    return result.docs.length > 0
      ? true
      : `${fieldLabel} must reference a published ${expectedType} system page.`
  }

```ts
import type { CollectionBeforeChangeHook } from 'payload'
import { APIError } from 'payload'

import {
  canPublishPages,
  canPublishSystemPages,
} from '@/access/adminCapabilities'
import type { EditorialWorkflowState } from '@/lib/editorialWorkflow'

type PublishCapability = 'publish_pages' | 'publish_system_pages'

type WorkflowDoc = {
  _status?: 'draft' | 'published'
  changeSummary?: string | null
  lastReviewedAt?: string | null
  lastReviewedBy?: number | null
  publishedAt?: string | null
  publishedBy?: number | null
  scheduledPublishAt?: string | null
  workflowState?: EditorialWorkflowState
}

const canPublish = (capability: PublishCapability, user: unknown) => {
  return capability === 'publish_pages'
    ? canPublishPages(user as never)
    : canPublishSystemPages(user as never)
}

export const enforceEditorialWorkflow =
  (publishCapability: PublishCapability): CollectionBeforeChangeHook<WorkflowDoc> =>
  async ({ data, originalDoc, req }) => {
    const nextState = (data?.workflowState ??
      originalDoc?.workflowState ??
      'draft') as EditorialWorkflowState
    const now = new Date().toISOString()
    const nextData = {
      ...data,
    }

    if (nextState === 'draft') {
      nextData._status = 'draft'
      return nextData
    }

    if (nextState === 'in_review') {
      nextData._status = 'draft'
      nextData.lastReviewedAt = now
      nextData.lastReviewedBy = typeof req.user?.id === 'number' ? req.user.id : null
      return nextData
    }

    if (!canPublish(publishCapability, req.user)) {
      throw new APIError('Only publishers and admins can move content to published.', 400)
    }

    if (nextState === 'scheduled') {
      if (!nextData.scheduledPublishAt) {
        throw new APIError('Scheduled content requires scheduledPublishAt.', 400)
      }

      nextData._status = 'draft'
      return nextData
    }

    if (nextState === 'published' || nextState === 'reverted') {
      nextData._status = 'published'
      nextData.publishedAt = now
      nextData.publishedBy = typeof req.user?.id === 'number' ? req.user.id : null
      nextData.scheduledPublishAt = null
      return nextData
    }

    return nextData
  }
```

import type { CollectionAfterChangeHook, GlobalAfterChangeHook } from 'payload'

import { recordOperationalAuditEvent } from '@/lib/operationalAudit'

type HookConfig<TDoc> = {
  entityType: string
  getSummary: (args: { doc: TDoc; previousDoc?: Partial<TDoc> | null }) => string
  titleField: keyof TDoc
}

export const createCollectionAuditHook = <TDoc extends { id?: number | string }>(
  config: HookConfig<TDoc>,
): CollectionAfterChangeHook<TDoc> => {
  return async ({ doc, operation, previousDoc, req }) => {
    await recordOperationalAuditEvent({
      action: operation,
      entityId: String(doc.id ?? ''),
      entityTitle: String(doc[config.titleField] ?? `#${doc.id}`),
      entityType: config.entityType,
      req,
      summary: config.getSummary({ doc, previousDoc }),
    })

    return doc
  }
}

export const createGlobalAuditHook = <TDoc extends Record<string, unknown>>(
  config: Omit<HookConfig<TDoc>, 'titleField'> & { title: string },
): GlobalAfterChangeHook<TDoc> => {
  return async ({ doc, previousDoc, req }) => {
    await recordOperationalAuditEvent({
      action: 'update',
      entityId: config.entityType,
      entityTitle: config.title,
      entityType: config.entityType,
      req,
      summary: config.getSummary({ doc, previousDoc }),
    })

    return doc
  }
}
